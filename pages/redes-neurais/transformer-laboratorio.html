Uma cabeça de atenção em JavaScript

O exemplo usa arrays e laços simples. As funções separam produto escalar, Softmax, atenção e previsão.

// Calcula a semelhança entre dois vetores.
function produtoEscalar(a, b) {
  var soma = 0;
  for (var i = 0; i < a.length; i++) soma = soma + a[i] * b[i];
  return soma;
}

// Converte escores em pesos positivos cuja soma vale 1.
function softmax(valores) {
  var resultado = [];
  var soma = 0;
  for (var i = 0; i < valores.length; i++) {
    resultado[i] = Math.exp(valores[i]);
    soma = soma + resultado[i];
  }
  for (i = 0; i < resultado.length; i++) resultado[i] = resultado[i] / soma;
  return resultado;
}

// A consulta é o último token; as chaves e valores são todo o prefixo.
function atencao(vetores) {
  var consulta = vetores[vetores.length - 1];
  var escores = [];
  for (var j = 0; j < vetores.length; j++) {
    escores[j] = produtoEscalar(consulta, vetores[j]) / Math.sqrt(3);
  }
  var pesos = softmax(escores);
  var contexto = [0, 0, 0];
  for (j = 0; j < vetores.length; j++) {
    for (var d = 0; d < 3; d++) contexto[d] = contexto[d] + pesos[j] * vetores[j][d];
  }
  return [contexto, pesos];
}

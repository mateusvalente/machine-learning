A rede será desenhada assim que os exemplos forem carregados.
1 · gerar(z)Calcula 49 somas ponderadas e aplica Sigmoid para formar os pixels falsos.
2 · discriminar(x)Soma os 49 pixels ponderados e aplica Sigmoid para estimar a chance de ser real.
3 · treinar DD compara real com alvo 1 e falso com alvo 0; somente seus pesos são ajustados.
4 · treinar GD fica congelado. O erro volta por D e corrige os pesos do Gerador.

Treinamento alternado da mini GAN

As funções do Gerador e do Discriminador ficam separadas para evidenciar qual rede é atualizada em cada fase.

gerar()discriminar()treinarDiscriminador()treinarGerador()
// Fase 1: D aprende que real = 1 e falso = 0.
function treinarDiscriminador(real, falso) {
  var dReal = discriminar(real);
  var dFalso = discriminar(falso);
  for (var i = 0; i < 49; i++) {
    var gradiente = ((dReal - 1) * real[i] + dFalso * falso[i]) / 2;
    pesosD[i] = pesosD[i] - taxaD * gradiente;
  }
}

// Fase 2: G tenta fazer D(G(z)) chegar a 1.
function treinarGerador(z) {
  var falso = gerar(z);
  var dFalso = discriminar(falso);
  for (var i = 0; i < 49; i++) {
    var gradientePixel = (dFalso - 1) * pesosD[i];
    var gradienteAtivacao = gradientePixel * falso[i] * (1 - falso[i]);
    for (var j = 0; j < 3; j++) {
      pesosG[i][j] = pesosG[i][j] - taxaG * gradienteAtivacao * z[j];
    }
  }
}

// ============================================================
// RANDOM FOREST DIDÁTICA: 2 entradas, 2 classes e árvores rasas
// Cada amostra contém [x1, x2, classe].
// ============================================================
var treino = [
  [1, 2, 'A'], [2, 3, 'A'], [2, 6, 'A'], [4, 7, 'A'],
  [6, 2, 'B'], [7, 4, 'B'], [8, 6, 'B'], [9, 8, 'B']
];

var teste = [
  [2, 4, 'A'], [3, 7, 'A'], [7, 3, 'B'], [8, 7, 'B']
];

// Sorteia N linhas com reposição. Uma linha pode aparecer mais de uma vez.
function bootstrap(dados) {
  var amostra = [];
  for (var i = 0; i < dados.length; i++) {
    var indice = Math.floor(Math.random() * dados.length);
    amostra[i] = dados[indice];
  }
  return amostra;
}

// Mede a impureza de uma lista de classes.
function gini(classes) {
  if (classes.length === 0) return 0;
  var quantidadeA = 0;
  for (var i = 0; i < classes.length; i++) {
    if (classes[i] === 'A') quantidadeA++;
  }
  var pA = quantidadeA / classes.length;
  var pB = 1 - pA;
  return 1 - pA * pA - pB * pB;
}

// Retorna a classe que aparece mais vezes.
function classeMajoritaria(classes) {
  var quantidadeA = 0;
  for (var i = 0; i < classes.length; i++) {
    if (classes[i] === 'A') quantidadeA++;
  }
  if (quantidadeA >= classes.length - quantidadeA) return 'A';
  return 'B';
}

// Treina uma árvore de um nível: [atributo, limiar, classeEsquerda, classeDireita].
function treinarArvore(dados) {
  var atributo = Math.floor(Math.random() * 2); // Random Forest sorteia atributos.
  var melhorGini = Infinity;
  var melhorArvore = null;

  for (var candidato = 0; candidato < dados.length; candidato++) {
    var limiar = dados[candidato][atributo];
    var esquerda = [];
    var direita = [];

    for (var i = 0; i < dados.length; i++) {
      if (dados[i][atributo] <= limiar) esquerda.push(dados[i][2]);
      else direita.push(dados[i][2]);
    }

    if (esquerda.length === 0 || direita.length === 0) continue;
    var impureza = (esquerda.length * gini(esquerda)
                  + direita.length * gini(direita)) / dados.length;

    if (impureza < melhorGini) {
      melhorGini = impureza;
      melhorArvore = [atributo, limiar,
        classeMajoritaria(esquerda), classeMajoritaria(direita)];
    }
  }
  return melhorArvore;
}

// Cada árvore recebe seu próprio bootstrap.
function treinarFloresta(dados, numeroArvores) {
  var floresta = [];
  while (floresta.length < numeroArvores) {
    var arvore = treinarArvore(bootstrap(dados));
    if (arvore !== null) floresta.push(arvore);
  }
  return floresta;
}

function preverArvore(arvore, entrada) {
  if (entrada[arvore[0]] <= arvore[1]) return arvore[2];
  return arvore[3];
}

// A previsão final é a classe com mais votos.
function prever(floresta, entrada) {
  var votosA = 0;
  for (var i = 0; i < floresta.length; i++) {
    if (preverArvore(floresta[i], entrada) === 'A') votosA++;
  }
  if (votosA > floresta.length / 2) return 'A';
  return 'B';
}

function avaliar(floresta, dados) {
  var acertos = 0;
  for (var i = 0; i < dados.length; i++) {
    if (prever(floresta, dados[i]) === dados[i][2]) acertos++;
  }
  return acertos / dados.length;
}

// ----------------------- USO DO MODELO -----------------------
var modelo = treinarFloresta(treino, 21);
console.log('Acurácia de teste:', avaliar(modelo, teste));
console.log('Previsão para [7, 5]:', prever(modelo, [7, 5]));
console.log('Primeira árvore [atributo, limiar, esquerda, direita]:', modelo[0]);

Por que não existe “treinar até convergir”?
O k-NN não ajusta pesos iterativamente. O resultado muda quando alteramos os dados, k, a métrica ou a regra de voto.

// ============================================================
// k-NN COMPLETO: 2 entradas, classificação binária e Min-Max
// Cada linha contém [x1, x2, classe].
// ============================================================
var treinoBruto = [
  [1, 15, 'A'], [2, 25, 'A'], [3, 35, 'A'], [4, 30, 'A'],
  [6, 60, 'B'], [7, 70, 'B'], [8, 65, 'B'], [9, 85, 'B']
];

var testeBruto = [
  [2, 20, 'A'], [4, 40, 'A'], [7, 55, 'B'], [8, 80, 'B']
];

var dadosTreino = [];
var minimos = [0, 0];
var maximos = [0, 0];

// Aprende a escala usando SOMENTE o treino.
function ajustarEscala(dados) {
  minimos = [dados[0][0], dados[0][1]];
  maximos = [dados[0][0], dados[0][1]];
  for (var i = 1; i < dados.length; i++) {
    for (var j = 0; j < 2; j++) {
      if (dados[i][j] < minimos[j]) minimos[j] = dados[i][j];
      if (dados[i][j] > maximos[j]) maximos[j] = dados[i][j];
    }
  }
}

function normalizarEntrada(entrada) {
  var normalizada = [];
  for (var j = 0; j < 2; j++) {
    normalizada[j] = (entrada[j] - minimos[j]) / (maximos[j] - minimos[j]);
  }
  return normalizada;
}

// No k-NN, treinar significa armazenar os exemplos preparados.
function treinar(dados) {
  ajustarEscala(dados);
  dadosTreino = [];
  for (var i = 0; i < dados.length; i++) {
    var x = normalizarEntrada(dados[i]);
    dadosTreino[i] = [x[0], x[1], dados[i][2]];
  }
}

function distanciaEuclidiana(a, b) {
  var dx1 = a[0] - b[0];
  var dx2 = a[1] - b[1];
  return Math.sqrt(dx1 * dx1 + dx2 * dx2);
}

// Ordenação simples para deixar o algoritmo visível aos alunos.
function ordenarPorDistancia(lista) {
  for (var volta = 0; volta < lista.length; volta++) {
    for (var i = 0; i < lista.length - 1; i++) {
      if (lista[i][0] > lista[i + 1][0]) {
        var auxiliar = lista[i];
        lista[i] = lista[i + 1];
        lista[i + 1] = auxiliar;
      }
    }
  }
}

function classificar(entradaBruta, k) {
  var entrada = normalizarEntrada(entradaBruta);
  var distancias = [];

  for (var i = 0; i < dadosTreino.length; i++) {
    distancias[i] = [
      distanciaEuclidiana(entrada, dadosTreino[i]),
      dadosTreino[i][2]
    ];
  }
  ordenarPorDistancia(distancias);

  var votosA = 0;
  for (var vizinho = 0; vizinho < k; vizinho++) {
    if (distancias[vizinho][1] === 'A') votosA++;
  }
  if (votosA > k / 2) return 'A';
  return 'B';
}

function avaliar(dados, k) {
  var acertos = 0;
  for (var i = 0; i < dados.length; i++) {
    if (classificar(dados[i], k) === dados[i][2]) acertos++;
  }
  return acertos / dados.length;
}

// ----------------------- USO DO MODELO -----------------------
treinar(treinoBruto);
console.log('Acurácia de teste:', avaliar(testeBruto, 3));
console.log('Previsão para [5, 48]:', classificar([5, 48], 3));
